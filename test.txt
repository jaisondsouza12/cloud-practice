function doPost(e) {
  var sheet=SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Sheet1");
  sheet.appendRow(
    [
      new Date(),
      e.parameter.name,
      e.parameter.email,
      e.parameter.course,
      e.parameter.phone,
      e.parameter.message
    ]
  );
  return ContentService.createTextOutput("Form Submitted Successfully!!!");
}
